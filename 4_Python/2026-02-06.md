## 總結

**Java 方式：**

- 重量級 IoC Container (Spring)
- 註解驅動 (@Autowired, @Component)
- 自動掃描與注入
- 適合大型企業應用

**Python 方式：**

- 輕量級依賴注入 (FastAPI Depends)
- 函數式 + Protocol
- 明確勝於隱式
- 適合快速開發與微服務

**我的建議：** 對於你的專案，使用 **FastAPI + Protocol + Depends** 就足夠了，不需要引入 `dependency-injector` 這類重量級套件。這更符合 Python 的「簡單勝於複雜」哲學，同時也能達到解耦和可測試性的目標。

## 1. **單例模式 (Singleton)**

### Java 方式 (複雜)

java

```java
public class Settings {
    private static Settings instance;
    private Settings() {}
    
    public static synchronized Settings getInstance() {
        if (instance == null) {
            instance = new Settings();
        }
        return instance;
    }
}
```

### Python 方式 (簡單)

python

```python
# 方式 1: 使用模組 (最 Pythonic)
# config.py
class Settings:
    DB_HOST = "localhost"

settings = Settings()  # 模組本身就是單例！

# 其他地方使用
from config import settings  # 永遠是同一個實例


# 方式 2: 使用裝飾器
from functools import lru_cache

@lru_cache(maxsize=1)
def get_settings():
    return Settings()


# 方式 3: 使用 __new__ (不推薦，太複雜)
class Settings:
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance
```

**Python 哲學：** 模組本身就是單例，不需要複雜的類別設計。

## 2. **工廠模式 (Factory)**

### Java 方式

java

```java
public interface Database {
    void connect();
}

public class DatabaseFactory {
    public static Database createDatabase(String type) {
        switch(type) {
            case "postgres": return new PostgresDB();
            case "mysql": return new MySQLDB();
            default: throw new IllegalArgumentException();
        }
    }
}
```

### Python 方式

python

```python
# 方式 1: 使用字典 (最簡單)
from typing import Protocol

class DatabaseProtocol(Protocol):
    def connect(self) -> None: ...

class PostgresDB:
    def connect(self): print("Connecting to Postgres")

class MySQLDB:
    def connect(self): print("Connecting to MySQL")

# 工廠就是字典！
DATABASE_FACTORIES = {
    'postgres': PostgresDB,
    'mysql': MySQLDB,
}

def create_database(db_type: str) -> DatabaseProtocol:
    return DATABASE_FACTORIES[db_type]()


# 方式 2: 使用函數
def create_database(db_type: str) -> DatabaseProtocol:
    if db_type == 'postgres':
        return PostgresDB()
    elif db_type == 'mysql':
        return MySQLDB()
    raise ValueError(f"Unknown database type: {db_type}")


# 方式 3: 使用類別方法 (接近 Java)
class DatabaseFactory:
    @staticmethod
    def create(db_type: str) -> DatabaseProtocol:
        match db_type:  # Python 3.10+
            case 'postgres': return PostgresDB()
            case 'mysql': return MySQLDB()
            case _: raise ValueError(f"Unknown: {db_type}")
```

**Python 哲學：** 函數和字典就夠了，不需要專門的工廠類別。

## 3. **策略模式 (Strategy)**

### Java 方式

java

```java
public interface PaymentStrategy {
    void pay(int amount);
}

public class CreditCardPayment implements PaymentStrategy {
    public void pay(int amount) { /* ... */ }
}

public class PayPalPayment implements PaymentStrategy {
    public void pay(int amount) { /* ... */ }

public class ShoppingCart {
    private PaymentStrategy paymentStrategy;
    
    public void setPaymentStrategy(PaymentStrategy strategy) {

}
```

### Python 方式

python

```python
# 方式 1: 使用函數 (First-class functions)
def credit_card_payment(amount: int):
    print(f"Paid {amount} with credit card")

def paypal_payment(amount: int):
    print(f"Paid {amount} with PayPal")

class ShoppingCart:
    def __init__(self, payment_strategy):
        self.payment_strategy = payment_strategy  # 直接傳入函數！
    
    def checkout(self, amount: int):
        self.payment_strategy(amount)

# 使用
cart = ShoppingCart(credit_card_payment)
cart.checkout(100)


# 方式 2: 使用 Protocol (需要狀態時)
from typing import Protocol

class PaymentStrategy(Protocol):
    def pay(self, amount: int) -> None: ...

class CreditCardPayment:
    def __init__(self, card_number: str):
        self.card_number = card_number
    
    def pay(self, amount: int):
        print(f"Charging {amount} to card {self.card_number}")

class ShoppingCart:
    def __init__(self, payment_strategy: PaymentStrategy):
        self.payment_strategy = payment_strategy
    
    def checkout(self, amount: int):
        self.payment_strategy.pay(amount)


# 方式 3: 使用字典 + 偏函數
from functools import partial

def payment(amount: int, method: str, **kwargs):
    if method == 'credit_card':
        print(f"Paid {amount} with card {kwargs['card_number']}")
    elif method == 'paypal':
        print(f"Paid {amount} with PayPal {kwargs['email']}")

# 創建策略
credit_payment = partial(payment, method='credit_card', card_number='1234')
paypal_payment = partial(payment, method='paypal', email='user@example.com')

credit_payment(100)
```

**Python 哲學：** 函數是一等公民，直接傳遞函數而不需要包裝成類別。

## 4. **裝飾器模式 (Decorator)**

### Java 方式

java

```java
public interface Coffee {
    double cost();
}

public class SimpleCoffee implements Coffee {
    public double cost() { return 5.0; }
}

public class MilkDecorator implements Coffee {
    private Coffee coffee;
    
    public MilkDecorator(Coffee coffee) {
        this.coffee = coffee;
    }
    
    public double cost() {
        return coffee.cost() + 1.5;
    }
}
```

### Python 方式

python

```python
# Python 內建裝飾器語法！
from functools import wraps
import time

def timer(func):
    """測量執行時間的裝飾器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} took {end-start:.2f}s")
        return result
    return wrapper

def retry(max_attempts=3):
    """重試裝飾器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"Retry {attempt + 1}/{max_attempts}")
        return wrapper
    return decorator

# 使用裝飾器
@timer
@retry(max_attempts=3)
def fetch_data():
    # 業務邏輯
    return "data"


# 類別裝飾器
def singleton(cls):
    instances = {}
    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    return get_instance

@singleton
class Settings:
    pass
```

**Python 哲學：** 語言原生支援裝飾器，不需要複雜的類別包裝。

## 5. **觀察者模式 (Observer)**

### Java 方式

java

```java
public interface Observer {
    void update(String message);
}

public class Subject {
    private List<Observer> observers = new ArrayList<>();
    
    public void attach(Observer observer) {
        observers.add(observer);
    }
    
    public void notifyObservers(String message) {
        for (Observer observer : observers) {
            observer.update(message);
        }
    }
}
```

### Python 方式

python

```python
# 方式 1: 使用簡單的列表和函數
class Subject:
    def __init__(self):
        self._observers = []
    
    def attach(self, callback):
        """直接註冊回調函數"""
        self._observers.append(callback)
    
    def notify(self, message):
        for callback in self._observers:
            callback(message)

# 使用
subject = Subject()
subject.attach(lambda msg: print(f"Observer 1: {msg}"))
subject.attach(lambda msg: print(f"Observer 2: {msg}"))
subject.notify("Event occurred!")


# 方式 2: 使用描述器 (Descriptor) - 屬性變化自動通知
class Observable:
    def __init__(self, name):
        self.name = name
        self.observers = []
    
    def __get__(self, obj, objtype=None):
        return getattr(obj, f'_{self.name}', None)
    
    def __set__(self, obj, value):
        setattr(obj, f'_{self.name}', value)
        for callback in self.observers:
            callback(value)

class Model:
    name = Observable('name')
    
    def observe_name(self, callback):
        self.name.observers.append(callback)

model = Model()
model.observe_name(lambda val: print(f"Name changed to: {val}"))
model.name = "Alice"  # 自動觸發通知


# 方式 3: 使用 asyncio (現代方式)
import asyncio

class AsyncSubject:
    def __init__(self):
        self._observers = []
    
    def attach(self, callback):
        self._observers.append(callback)
    
    async def notify(self, message):
        tasks = [callback(message) for callback in self._observers]
        await asyncio.gather(*tasks)

async def observer1(msg):
    print(f"Observer 1: {msg}")

async def observer2(msg):
    await asyncio.sleep(1)
    print(f"Observer 2: {msg}")

# 使用
subject = AsyncSubject()
subject.attach(observer1)
subject.attach(observer2)
await subject.notify("Async event!")
```

**Python 哲學：** 回調函數、lambda、asyncio 讓觀察者模式更簡潔。

# 方式 1: FastAPI 依賴注入 (推薦給 Web 應用)
from typing import Annotated, Protocol
from fastapi import Depends

class SettingsProtocol(Protocol):
    DB_HOST: str
    DEBUG: bool

def get_settings() -> SettingsProtocol:
    return Settings()

# 使用 Annotated 簡化
SettingsDep = Annotated[SettingsProtocol, Depends(get_settings)]

@app.get("/")
def endpoint(settings: SettingsDep):
    return {"host": settings.DB_HOST}


# 方式 2: 上下文管理器
from contextlib import contextmanager

@contextmanager
def inject_settings():
    settings = Settings()
    try:
        yield settings
    finally:
        # 清理邏輯
        pass

with inject_settings() as settings:
    db = Database(settings)


# 方式 3: 函數式編程 (最簡單)
def create_app(settings: SettingsProtocol):
    """工廠函數接收依賴"""
    def get_db():
        return Database(settings)
    
    def get_cache():
        return Cache(settings)
    
    return {
        'db': get_db,
        'cache': get_cache
    }

# 使用
settings = get_settings()
app = create_app(settings)
db = app['db']()

## Python 設計模式總結

|模式|Java 方式|Python 方式|Python 特性|
|---|---|---|---|
|**Singleton**|複雜的類別實作|模組/lru_cache|模組即單例|
|**Factory**|工廠類別|字典/函數|First-class functions|
|**Strategy**|策略介面+實作類別|直接傳函數|函數是物件|
|**Decorator**|裝飾器類別|@decorator|語法原生支援|
|**Observer**|Observer 介面|回調列表/asyncio|動態類型|
|**Builder**|Builder 類別|字典/**kwargs|動態參數|
|**Template Method**|抽象類別|繼承/組合|Duck typing|
|**DI/IoC**|Spring Container|Depends/函數|明確勝於隱式|

## Python 設計原則 (Pythonic Design Principles)

### 1. **EAFP > LBYL**

python

```python
# Easier to Ask Forgiveness than Permission
try:
    value = my_dict['key']
except KeyError:
    value = default

# NOT: Look Before You Leap
if 'key' in my_dict:
    value = my_dict['key']
else:
    value = default
```

### 2. **Duck Typing > 嚴格型別**

python

```python
# 如果它像鴨子，就當它是鴨子
def process(file_like):
    """接受任何有 read() 方法的物件"""
    return file_like.read()

# 可以傳入: 檔案、StringIO、網路流等
```

### 3. **Protocol > Interface**

python

```python
from typing import Protocol

class Drawable(Protocol):
    def draw(self) -> None: ...

# 不需要繼承，只要有 draw() 就符合
```

### 4. **組合 > 繼承**

python

```python
# 使用組合
class DatabaseService:
    def __init__(self, settings, logger):
        self.settings = settings
        self.logger = logger

# 而非深層繼承
class DatabaseService(BaseService, LoggableMixin, ConfigurableMixin):
    pass
```

# 1. Protocol 定義介面
from typing import Protocol

class SettingsProtocol(Protocol):
    DB_HOST: str
    DB_CONNECT_STRING: str
    DEBUG: bool

class LoggerProtocol(Protocol):
    def info(self, msg: str) -> None: ...
    def error(self, msg: str) -> None: ...


# 2. 簡單的工廠函數
def create_database_service(
    settings: SettingsProtocol,
    logger: LoggerProtocol
):
    """工廠函數 - 比工廠類別更 Pythonic"""
    engine = create_async_engine(
        settings.DB_CONNECT_STRING,
        echo=settings.DEBUG
    )
    
    return {
        'engine': engine,
        'get_session': lambda: sessionmaker(bind=engine)(),
        'logger': logger
    }


# 3. 裝飾器增強功能
from functools import wraps

def with_logging(logger: LoggerProtocol):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            logger.info(f"Calling {func.__name__}")
            try:
                result = await func(*args, **kwargs)
                logger.info(f"{func.__name__} succeeded")
                return result
            except Exception as e:
                logger.error(f"{func.__name__} failed: {e}")
                raise
        return wrapper
    return decorator


# 4. FastAPI 依賴注入
from typing import Annotated
from fastapi import Depends

SettingsDep = Annotated[SettingsProtocol, Depends(get_settings)]
LoggerDep = Annotated[LoggerProtocol, Depends(get_logger)]

@with_logging(logger)
@app.get("/users")
async def get_users(
    settings: SettingsDep,
    logger: LoggerDep,
    db: Annotated[AsyncSession, Depends(get_async_db)]
):
    logger.info(f"Fetching users from {settings.DB_HOST}")
    return await db.execute(select(User))
